# ============================================================================= # COMPILER FLAGS CONFIGURATION # ============================================================================= # This module provides compiler-specific flags and optimizations for the CryptoGL library. # It uses CMake's generator expressions to apply different flags based on the compiler. # Function to set compiler flags for a specific target # Usage: set_compiler_flags(your_target_name) function(set_compiler_flags target) # ============================================================================= # C++ STANDARD ENFORCEMENT # ============================================================================= # Ensure the target uses C++20 standard features target_compile_features(${target} PRIVATE cxx_std_20) # ============================================================================= # WARNING FLAGS (Code Quality) # ============================================================================= # Enable comprehensive warnings to catch potential issues early target_compile_options(${target} PRIVATE # GCC and Clang: Enable all warnings and treat them as errors $<$:-Wall -Wextra -Wpedantic -Werror> # MSVC: Enable all warnings and treat them as errors $<$:/W4 /WX> ) # ============================================================================= # SECURITY FLAGS (Memory Protection) # ============================================================================= # Enable security features to prevent common attack vectors target_compile_options(${target} PRIVATE # GCC and Clang security flags: # -fstack-protector-strong: Stack canary protection # -fPIE: Position Independent Executable (ASLR support) # -fPIC: Position Independent Code (for shared libraries) $<$:-fstack-protector-strong -fPIE -fPIC> # MSVC security flags: # /GS: Buffer security check # /DYNAMICBASE: Enable ASLR # /NXCOMPAT: Enable DEP (Data Execution Prevention) $<$:/GS /DYNAMICBASE /NXCOMPAT> ) # ============================================================================= # PERFORMANCE OPTIMIZATION FLAGS # ============================================================================= # Enable aggressive optimizations for maximum performance target_compile_options(${target} PRIVATE # GCC and Clang performance flags: # -O3: Maximum optimization level # -march=native: Optimize for current CPU architecture # -mtune=native: Tune for current CPU microarchitecture $<$:-O3 -march=native -mtune=native> # MSVC performance flags: # /O2: Maximum optimization # /GL: Whole program optimization # /arch:AVX2: Enable AVX2 instructions $<$:/O2 /GL /arch:AVX2> ) # ============================================================================= # CRYPTO-SPECIFIC OPTIMIZATIONS # ============================================================================= # Special optimizations beneficial for cryptographic operations target_compile_options(${target} PRIVATE # GCC and Clang crypto optimizations: # -fomit-frame-pointer: Save registers for better performance # -funroll-loops: Unroll loops for better instruction-level parallelism $<$:-fomit-frame-pointer -funroll-loops> # MSVC crypto optimizations: # /Oy: Omit frame pointer # /Ob2: Any suitable inline function expansion $<$:/Oy /Ob2> ) endfunction()

2.2.3 cmake/SecurityHardening.cmake

# ============================================================================= # SECURITY HARDENING CONFIGURATION # ============================================================================= # This module applies security hardening measures to protect against common # attack vectors like buffer overflows, memory corruption, and code injection. # Function to apply security hardening to a specific target # Usage: apply_security_hardening(your_target_name) function(apply_security_hardening target) # ============================================================================= # MEMORY PROTECTION DEFINITIONS # ============================================================================= # Enable runtime checks and protections at the source level target_compile_definitions(${target} PRIVATE _FORTIFY_SOURCE=2 # Enable buffer overflow detection _GLIBCXX_ASSERTIONS # Enable C++ standard library assertions ) # ============================================================================= # LINKER SECURITY FLAGS # ============================================================================= # Apply security flags during the linking phase target_link_options(${target} PRIVATE # GCC and Clang linker security flags: # -Wl,-z,relro: Read-only relocations (prevents GOT overwrites) # -Wl,-z,now: Immediate binding (prevents lazy binding attacks) # -Wl,-z,noexecstack: Non-executable stack (prevents stack-based attacks) $<$:-Wl,-z,relro -Wl,-z,now -Wl,-z,noexecstack> # MSVC linker security flags: # /NXCOMPAT: Non-executable memory compatibility # /DYNAMICBASE: Enable ASLR (Address Space Layout Randomization) # /HIGHENTROPYVA: High entropy address space layout $<$:/NXCOMPAT /DYNAMICBASE /HIGHENTROPYVA> ) # ============================================================================= # SECURITY MODE CONFIGURATION # ============================================================================= # Disable potentially dangerous features in production builds target_compile_definitions(${target} PRIVATE NDEBUG # Disable debug features in release builds CRYPTOGL_SECURE_MODE=1 # Enable secure mode for the library ) # ============================================================================= # RUNTIME SECURITY CHECKS # ============================================================================= # Enable sanitizers for development builds to catch security issues target_compile_options(${target} PRIVATE # GCC and Clang sanitizers: # -fsanitize=address: Address sanitizer (detects memory errors) # -fsanitize=undefined: Undefined behavior sanitizer # -fno-omit-frame-pointer: Keep frame pointers for better debugging $<$:-fsanitize=address,undefined -fno-omit-frame-pointer> ) endfunction() # ============================================================================= # SECURITY TESTING CONFIGURATION # ============================================================================= # Option to enable security-specific tests (penetration testing, fuzzing, etc.) option(ENABLE_SECURITY_TESTS "Enable security-specific tests" ON) # If security tests are enabled, include the security test directory if(ENABLE_SECURITY_TESTS) add_subdirectory(tests/security) endif()

2.2.4 cmake/PerformanceOptimization.cmake

# ============================================================================= # PERFORMANCE OPTIMIZATION CONFIGURATION # ============================================================================= # This module applies advanced performance optimizations including SIMD, # Link Time Optimization (LTO), and Profile-Guided Optimization (PGO). # Function to apply performance optimizations to a specific target # Usage: apply_performance_optimization(your_target_name) function(apply_performance_optimization target) # ============================================================================= # SIMD OPTIMIZATIONS # ============================================================================= # Enable SIMD (Single Instruction, Multiple Data) instructions for # parallel processing of data, crucial for cryptographic operations include(CheckCXXCompilerFlag) # Check for AVX2 support (256-bit vector instructions) # AVX2 provides the best performance for cryptographic algorithms check_cxx_compiler_flag("-mavx2" COMPILER_SUPPORTS_AVX2) if(COMPILER_SUPPORTS_AVX2) target_compile_options(${target} PRIVATE -mavx2) # Enable AVX2 instructions target_compile_definitions(${target} PRIVATE HAVE_AVX2=1) # Define preprocessor macro message(STATUS "AVX2 support enabled for ${target}") else() message(STATUS "AVX2 not supported by compiler for ${target}") endif() # Check for SSE4.2 support (128-bit vector instructions) # SSE4.2 is a fallback for older processors check_cxx_compiler_flag("-msse4.2" COMPILER_SUPPORTS_SSE42) if(COMPILER_SUPPORTS_SSE42) target_compile_options(${target} PRIVATE -msse4.2) # Enable SSE4.2 instructions target_compile_definitions(${target} PRIVATE HAVE_SSE4_2=1) # Define preprocessor macro message(STATUS "SSE4.2 support enabled for ${target}") else() message(STATUS "SSE4.2 not supported by compiler for ${target}") endif() # ============================================================================= # LINK TIME OPTIMIZATION (LTO) # ============================================================================= # LTO allows the compiler to optimize across multiple source files # This can provide significant performance improvements (5-15%) include(CheckIPOSupported) check_ipo_supported(RESULT ipo_supported OUTPUT ipo_output) if(ipo_supported) set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION TRUE # Enable LTO for this target ) message(STATUS "Link Time Optimization enabled for ${target}") else() message(WARNING "LTO not supported for ${target}: ${ipo_output}") endif() # ============================================================================= # PROFILE-GUIDED OPTIMIZATION (PGO) # ============================================================================= # PGO uses runtime profiling data to optimize the code for actual usage patterns # This can provide 10-20% performance improvement but requires two build phases option(ENABLE_PGO "Enable Profile-Guided Optimization" OFF) if(ENABLE_PGO) # Phase 1: Generate profiling data set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fprofile-generate") set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -fprofile-generate") message(STATUS "Profile-Guided Optimization enabled for ${target}") message(STATUS "Note: PGO requires two build phases - first with -fprofile-generate,") message(STATUS "then run your application, then rebuild with -fprofile-use") endif() endfunction()

🔒 3. Security Considerations

Critical Security Notes:
  • Never hardcode cryptographic keys in source code
  • Use secure random number generators for all cryptographic operations
  • Implement proper memory wiping for sensitive data
  • Enable all security hardening flags in production builds
  • Regular security audits of the codebase are essential

3.1 Security Hardening Checklist

  • ✅ Stack protection enabled
  • ✅ Position Independent Executable (PIE)
  • ✅ Address Space Layout Randomization (ASLR)
  • ✅ Non-executable stack
  • ✅ Read-only relocations
  • ✅ Immediate binding
  • ✅ Undefined behavior sanitizer
  • ✅ Address sanitizer

Level Flags Use Case Performance Impact

  2. SSE4.2 (128-bit vectors) - Good performance
  3. Fallback to scalar operations - Compatible
