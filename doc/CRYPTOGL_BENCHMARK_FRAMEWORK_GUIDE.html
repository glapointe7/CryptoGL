🎯 Framework Overview

Design Principles

  • Constant-Time Focus: Resistant to timing attacks with statistical analysis
  • Security-First: Memory-safe, side-channel resistant measurements
  • High Performance: Minimal overhead, optimized measurement loops
  • Readable Results: Clear visualization and statistical reporting
  • Easy Integration: Seamless extension of existing Test.hpp framework

⚠️ Security Considerations

This benchmark framework is specifically designed for cryptographic code where timing consistency is critical. It includes features to detect timing variations that could lead to side-channel vulnerabilities.

    /**
     * @brief High-precision timing measurement for cryptographic operations
     * 
     * Provides nanosecond-precision timing with CPU fence instructions
     * to prevent instruction reordering and ensure accurate measurements.
     */
    class TimingMeasurement
    {
    public:
        using TimePoint = std::chrono::high_resolution_clock::time_point;
        using Duration = std::chrono::nanoseconds;
        
        /**
         * @brief Start high-precision timing measurement
         * @note Uses CPU serialization instructions to prevent reordering
         */
        static void startMeasurement() noexcept
        {
            #if defined(__x86_64__) || defined(_M_X64)
            _mm_lfence(); // Load fence - wait for all previous loads
            _mm_mfence(); // Memory fence - full barrier
            #endif
            std::atomic_thread_fence(std::memory_order_seq_cst);
        }
        
        /**
         * @brief End timing measurement with serialization
         * @return Duration in nanoseconds
         */
        static Duration endMeasurement(TimePoint start) noexcept
        {
            #if defined(__x86_64__) || defined(_M_X64)
            _mm_mfence(); // Ensure all operations completed
            #endif
            std::atomic_thread_fence(std::memory_order_seq_cst);
            return std::chrono::high_resolution_clock::now() - start;
        }
        
        /**
         * @brief Get current high-resolution timestamp
         */
        static TimePoint now() noexcept
        {
            return std::chrono::high_resolution_clock::now();
        }
    };

    /**
     * @brief Statistical analysis for benchmark measurements
     * 
     * Provides comprehensive statistical analysis including detection
     * of timing anomalies that could indicate side-channel vulnerabilities.
     */
    class BenchmarkStatistics
    {
    public:
        /**
         * @brief Statistical results for benchmark analysis
         */
        struct Statistics
        {
            double mean_ns;              // Mean execution time in nanoseconds
            double median_ns;            // Median execution time
            double min_ns;               // Minimum execution time
            double max_ns;               // Maximum execution time
            double stddev_ns;           // Standard deviation
            double cv_percent;          // Coefficient of variation as percentage
            double percentile_95_ns;    // 95th percentile
            double percentile_99_ns;    // 99th percentile
            bool is_constant_time;       // True if CV < threshold
            size_t sample_count;        // Number of measurements
        };
        
        /**
         * @brief Calculate comprehensive statistics from timing measurements
         * @param measurements Vector of timing measurements in nanoseconds
         * @param constant_time_threshold CV threshold for constant-time classification (default 5%)
         * @return Complete statistical analysis
         */
        static Statistics calculate(const Vector<double>& measurements,
                                         double constant_time_threshold = 5.0)
        {
            if (measurements.empty())
            {
                return Statistics{0, 0, 0, 0, 0, 0, 0, 0, false, 0};
            }
            
            // Create sorted copy for percentile calculations
            auto sorted = measurements;
            std::sort(sorted.begin(), sorted.end());
            
            // Basic statistics
            double sum = std::accumulate(measurements.begin(), measurements.end(), 0.0);
            double mean = sum / measurements.size();
            
            // Variance and standard deviation
            double variance = 0.0;
            for (double value : measurements)
            {
                double diff = value - mean;
                variance += diff * diff;
            }
            variance /= measurements.size();
            double stddev = std::sqrt(variance);
            
            // Coefficient of variation (CV)
            double cv = (mean > 0) ? (stddev / mean * 100.0) : 0.0;
            
            // Percentiles
            double median = getPercentile(sorted, 50.0);
            double p95 = getPercentile(sorted, 95.0);
            double p99 = getPercentile(sorted, 99.0);
            
            return Statistics
            {
                mean,
                median,
                sorted.front(),
                sorted.back(),
                stddev,
                cv,
                p95,
                p99,
                cv < constant_time_threshold,
                measurements.size()
            };
        }
        
    private:
        /**
         * @brief Calculate percentile from sorted data
         */
        static double getPercentile(const Vector<double>& sorted_data, double percentile)
        {
            if (sorted_data.empty()) return 0.0;
            
            double index = percentile / 100.0 * (sorted_data.size() - 1);
            size_t lower_index = static_cast<size_t>(index);
            size_t upper_index = std::min(lower_index + 1, sorted_data.size() - 1);
            
            double weight = index - lower_index;
            return sorted_data[lower_index] * (1.0 - weight) + sorted_data[upper_index] * weight;
        }
    };

    /**
     * @brief Advanced benchmark framework derived from Test.hpp
     * 
     * Extends the base Test class with comprehensive benchmarking capabilities
     * specifically designed for cryptographic applications requiring constant-time
     * execution and side-channel resistance.
     */
    class Benchmark : public Test
    {
    public:
        /**
         * @brief Configuration for benchmark execution
         */
        struct Config
        {
            size_t warmup_iterations = 1000;       // Warm up CPU/cache
            size_t benchmark_iterations = 10000;   // Main benchmark runs
            double constant_time_threshold = 5.0;     // CV% threshold for constant-time
            bool enable_memory_tracking = true;       // Track memory usage
            bool enable_statistical_analysis = true; // Full statistical output
            bool randomize_input_data = true;        // Use random inputs to prevent branch prediction
        };
        
        /**
         * @brief Initialize benchmark with configuration
         */
        explicit Benchmark(const Config& config = Config{}) : m_config(config) {}
        
        /**
         * @brief Execute benchmark with statistical analysis
         * @param benchmark_function Lambda or function pointer to benchmark
         * @return Complete statistical analysis of execution times
         */
        template<typename BenchmarkFunc>
        BenchmarkStatistics::Statistics execute(BenchmarkFunc&& benchmark_function)
        {
            // Set CPU affinity and priority for consistent measurements
            setupCPUAffinity();
            
            // Storage for timing measurements
            Vector<double> measurements;
            measurements.reserve(m_config.benchmark_iterations);
            
            // Warmup phase to stabilize CPU frequency and cache
            warmupPhase(benchmark_function);
            
            // Main benchmark loop
            for (size_t i = 0; i < m_config.benchmark_iterations; ++i)
            {
                // Optional input randomization to prevent branch prediction optimizations
                if (m_config.randomize_input_data)
                {
                    randomizeInputData();
                }
                
                // High-precision timing measurement
                TimingMeasurement::startMeasurement();
                auto start = TimingMeasurement::now();
                
                // Execute the benchmark function
                benchmark_function();
                
                // End measurement with memory barriers
                auto duration = TimingMeasurement::endMeasurement(start);
                
                // Store timing in nanoseconds
                measurements.push_back(static_cast<double>(duration.count()));
            }
            
            // Calculate comprehensive statistics
            auto stats = BenchmarkStatistics::calculate(measurements, m_config.constant_time_threshold);
            
            // Store for reporting
            m_last_statistics = stats;
            
            return stats;
        }
        
        /**
         * @brief Print comprehensive benchmark results
         */
        void printResult() const override
        {
            printBenchmarkResults(m_last_statistics);
        }
        
        /**
         * @brief Macro-compatible benchmark execution
         * @param benchmark_lambda Lambda function to benchmark
         */
        void runBenchmark(std::function<void()> benchmark_lambda)
        {
            auto stats = execute(benchmark_lambda);
            has_passed = stats.is_constant_time || !m_require_constant_time;
        }
        
        /**
         * @brief Set whether constant-time execution is required for test to pass
         */
        void requireConstantTime(bool require = true) { m_require_constant_time = require; }
        
    protected:
        /**
         * @brief Default setUp implementation - can be overridden
         */
        void setUp() override {}
        
        /**
         * @brief Default run implementation - should be overridden
         */
        void run() override {}
        
        /**
         * @brief Default tearDown implementation - can be overridden
         */
        void tearDown() override {}
        
    private:
        Config m_config;
        BenchmarkStatistics::Statistics m_last_statistics;
        bool m_require_constant_time = false;
        std::random_device m_random_device;
        std::mt19937_64 m_rng{m_random_device()};
        
        // Implementation methods would continue here...
        // [Additional private methods for CPU affinity, warmup, etc.]
    };
}

#ifdef _WIN32
    #include <windows.h>

#endif

namespace UnitTests
{
    /**
     * @brief Set CPU affinity to a specific core for consistent measurements
     * 
     * Binds the current thread to a single CPU core to reduce timing variance
     * caused by thread migration between cores with different cache states.
     */
    void Benchmark::setupCPUAffinity()
    {
#ifdef _WIN32
        // Windows implementation
        HANDLE currentThread = GetCurrentThread();
        DWORD_PTR affinityMask = 1; // Bind to CPU 0
        SetThreadAffinityMask(currentThread, affinityMask);
        
        // Set high priority for timing accuracy
        SetThreadPriority(currentThread, THREAD_PRIORITY_TIME_CRITICAL);
#else
        // Linux implementation
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        CPU_SET(0, &cpuset); // Bind to CPU 0
        
        pthread_t currentThread = pthread_self();
        pthread_setaffinity_np(currentThread, sizeof(cpu_set_t), &cpuset);
        
        // Set high priority
        struct sched_param param;
        param.sched_priority = sched_get_priority_max(SCHED_FIFO);
        pthread_setschedparam(currentThread, SCHED_FIFO, &param);
#endif
    }

    /**
     * @brief Warmup phase to stabilize CPU frequency and cache state
     * @param benchmark_function Function to execute during warmup
     */
    template<typename BenchmarkFunc>
    void Benchmark::warmupPhase(BenchmarkFunc&& benchmark_function)
    {
        // Execute warmup iterations to stabilize performance
        for (size_t i = 0; i < m_config.warmup_iterations; ++i)
        {
            benchmark_function();
            
            // Prevent compiler from optimizing away the function call
            std::atomic_thread_fence(std::memory_order_seq_cst);
        }
        
        // Brief pause to let CPU stabilize
        std::this_thread::sleep_for(std::chrono::milliseconds(10));
    }

    /**
     * @brief Randomize input data to prevent branch prediction optimizations
     */
    void Benchmark::randomizeInputData()
    {
        // Generate random memory access pattern
        static thread_local std::uniform_int_distribution<uint64_t> dist;
        volatile uint64_t dummy = dist(m_rng);
        // This prevents compiler from optimizing based on predictable inputs
    }

    /**
     * @brief Print comprehensive benchmark results with security analysis
     * @param stats Statistical analysis results
     */
    void Benchmark::printBenchmarkResults(const BenchmarkStatistics::Statistics& stats) const
    {
        // Color-coded output based on security analysis
        std::string status_color = stats.is_constant_time 
            ? Color::FG_GREEN 
            : Color::FG_YELLOW;
        
        std::cout << status_color << "\n┌─ BENCHMARK RESULTS ─────────────────────────────────────┐" << Color::FG_DEFAULT << "\n";
        std::cout << "│ Test: " << Color::FG_CYAN << getName() << Color::FG_DEFAULT;
        
        // Performance metrics
        std::cout << std::fixed << std::setprecision(2);
        std::cout << "\n│ Mean Time:     " << Color::FG_WHITE << stats.mean_ns << " ns" << Color::FG_DEFAULT;
        std::cout << "\n│ Median Time:   " << stats.median_ns << " ns";
        std::cout << "\n│ Min Time:      " << stats.min_ns << " ns";
        std::cout << "\n│ Max Time:      " << stats.max_ns << " ns";
        
        // Statistical analysis
        std::cout << "\n│ Std Deviation: " << stats.stddev_ns << " ns";
        std::cout << "\n│ 95th Percentile: " << stats.percentile_95_ns << " ns";
        std::cout << "\n│ 99th Percentile: " << stats.percentile_99_ns << " ns";
        
        // Security analysis - most important for crypto
        std::cout << std::setprecision(3);
        std::cout << "\n│ CV (Variation): " << stats.cv_percent << "%";
        
        std::string security_status = stats.is_constant_time 
            ? "✓ CONSTANT-TIME" 
            : "⚠ VARIABLE-TIME";
        std::string security_color = stats.is_constant_time 
            ? Color::FG_GREEN 
            : Color::FG_RED;
        
        std::cout << "\n│ Security:      " << security_color << security_status << Color::FG_DEFAULT;
        std::cout << "\n│ Samples:       " << stats.sample_count;
        
        std::cout << status_color << "\n└─────────────────────────────────────────────────────────┘" << Color::FG_DEFAULT << "\n";
        
        // Security recommendations if variable-time detected
        if (!stats.is_constant_time)
        {
            std::cout << Color::FG_YELLOW << "⚠  WARNING: Variable timing detected (CV > 5%)\n"
                      << "   This may indicate potential timing attack vulnerability.\n"
                      << "   Consider implementing constant-time algorithms.\n" << Color::FG_DEFAULT;
        }
    }

2. Benchmark Macro Definition

/**
 * @brief Macro for easy benchmark test creation
 * 
 * Usage: BENCHMARK(MyBenchmarkTest, Benchmark) { ... }
 * This creates a benchmark test that inherits from the Benchmark class
 */
#define BENCHMARK(ClassName, Parent) \
    class ClassName : public Parent \
    { \
    public: \
        ClassName() { setName(#ClassName); } \
        void run() override; \
    }; \
    class Add##ClassName \
    { \
    public: \
        Add##ClassName() { TestContainer::getInstance().append(new ClassName()); } \
    }; \
    static Add##ClassName AddThis##ClassName; \
    void ClassName::run()

/**
 * @brief Simplified benchmark execution macro
 * 
 * Usage within benchmark test:
 * MEASURE([&](){ myFunction(input); });
 */
#define MEASURE(lambda_function) \
    runBenchmark(lambda_function)

1. Basic Cryptographic Function Benchmark

#include "Benchmark.hpp"
#include "../../src/algorithms/block_cipher/AES/AES.hpp"

/**
 * @brief Benchmark AES-256 encryption for constant-time validation
 * 
 * This test verifies that AES encryption runs in constant time
 * regardless of input data, which is crucial for side-channel resistance.
 */
BENCHMARK(AES256EncryptionBenchmark, Benchmark)
{
    // Test setup
    AES aes;
    Vector<uint8_t> key(32, 0x42); // 256-bit key
    Vector<uint8_t> plaintext(16, 0x00); // 128-bit block
    Vector<uint8_t> ciphertext(16);
    
    aes.setKey(key);
    
    // Require constant-time execution (critical for AES)
    requireConstantTime(true);
    
    // Benchmark the encryption operation
    MEASURE([&]() {
        aes.encryptBlock(plaintext, ciphertext);
    });
}

3. Memory Access Pattern Security Test

/**
 * @brief Test memory access patterns for cache timing resistance
 * 
 * Validates that cryptographic operations don't leak information
 * through cache access patterns that could enable cache timing attacks.
 */
BENCHMARK(CacheTimingResistanceBenchmark, Benchmark)
{
    // Large lookup table to test cache behavior
    constexpr size_t TABLE_SIZE = 4096;
    Vector<uint8_t> lookup_table(TABLE_SIZE);
    std::iota(lookup_table.begin(), lookup_table.end(), 0);
    
    // Test various access patterns
    Vector<uint16_t> indices = {
        0, 1, 255, 256, 1023, 1024, 2047, 4095
    };
    
    size_t index_counter = 0;
    
    // Very strict constant-time requirement for cache resistance
    requireConstantTime(true);
    
    MEASURE([&]() {
        // Access different indices each iteration
        size_t index = indices[index_counter % indices.size()];
        volatile uint8_t value = lookup_table[index];
        ++index_counter;
        
        // Prevent compiler from optimizing away the access
        __asm__ volatile("" : : "r"(value) : "memory");
    });
}

/**

 * Demonstrates how to benchmark multiple implementations and compare

    double speedup = sha3_stats.mean_ns / sha256_stats.mean_ns;
    std::cout << "Speedup: " << std::fixed << std::setprecision(2) << speedup << "x\n";
    
    // Test passes if both are reasonably constant-time
    setPassed(sha256_stats.is_constant_time && sha3_stats.is_constant_time);
}
